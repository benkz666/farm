模拟丢包恢复

```javascript
(() => {
  const DROP_COUNT = 3 // 修改这里：连续丢弃多少次操作

  const client = window.__farm?.getNetClient?.()
  if (!client) throw new Error('请先登录并进入农场')

  window.__stopGapDemo?.()

  const originalDispatch = client._dispatchPush.bind(client)
  const originalSyncFarm = client.syncFarm.bind(client)

  let droppedCount = 0
  let recoveryTriggered = false
  const droppedSequences = []

  const demoDispatch = (env) => {
    if (env?.cmd === 9000) {
      const seq = env.payload?.farm_seq

      if (droppedCount < DROP_COUNT) {
        droppedCount++
        droppedSequences.push(seq)

        console.warn(
          `%c[缺口演示] 丢弃 ${droppedCount}/${DROP_COUNT}：FarmDelta seq=${seq}`,
          'color:#d97706;font-weight:bold',
        )

        if (droppedCount === DROP_COUNT) {
          console.warn(
            `%c[缺口演示] 已连续丢弃 ${DROP_COUNT} 条，请再执行 1 次有效操作触发恢复`,
            'color:#dc2626;font-weight:bold',
          )
        }

        return
      }

      if (!recoveryTriggered) {
        recoveryTriggered = true
        console.warn(
          `%c[缺口演示] 收到后续 FarmDelta seq=${seq}，开始检测消息缺口`,
          'color:#dc2626;font-weight:bold',
        )
      }
    }

    return originalDispatch(env)
  }

  const demoSyncFarm = async (ownerUid, fromSeq) => {
    console.warn(
      `%c[缺口演示] 自动请求 SyncFarm：owner=${ownerUid}，from_seq=${fromSeq}`,
      'color:#2563eb;font-weight:bold',
    )

    const result = await originalSyncFarm(ownerUid, fromSeq)
    const payload = result?.payload || {}
    const recoveredSequences = Array.isArray(payload.deltas)
      ? payload.deltas.map(delta => delta.farm_seq)
      : []

    console.warn(
      `%c[缺口演示] 恢复完成
故意丢弃：${droppedSequences.join(', ')}
服务端补回：${recoveredSequences.join(', ') || '全量快照'}
最终序列：${payload.farm_seq}
恢复方式：${payload.snapshot ? '全量快照' : '连续增量'}`,
      'color:#059669;font-weight:bold',
    )

    return result
  }

  client._dispatchPush = demoDispatch
  client.syncFarm = demoSyncFarm

  window.__stopGapDemo = () => {
    if (client._dispatchPush === demoDispatch) {
      client._dispatchPush = originalDispatch
    }
    if (client.syncFarm === demoSyncFarm) {
      client.syncFarm = originalSyncFarm
    }

    delete window.__stopGapDemo
    console.log('[缺口演示] 已关闭')
  }

  console.log(
    `%c[缺口演示] 已启动：将连续丢弃 ${DROP_COUNT} 条 FarmDelta`,
    'color:#059669;font-weight:bold',
  )
})()
```



模拟断网恢复

网络改成Offline， B 在 A 的农场执行几次操作，然后把网络改成No throttling

```javascript
(() => {
  const client = window.__farm?.getNetClient?.()
  if (!client) throw new Error('请先登录并进入农场')

  client._detachSocket()
  client._failAllPending(new Error('demo: simulated network interruption'))
  client._scheduleReconnect()

  console.warn('[断线演示] WebSocket 已断开，客户端正在自动重连')
})()
```
